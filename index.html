高頻容易遺忘的持久化與驗證指令
1 · SELinux 永久標籤與套用 
semanage fcontext -a -t httpd_sys_content_t '/web(/.*)?'
restorecon -Rv /web
2 · 修改 fstab 後強制驗證 
mount -a
findmnt --verify
lsblk -f
3 · nmcli 靜態網路設定與啟用 
nmcli con add con-name lab type ethernet ifname enp1s0 \
  ipv4.method manual ipv4.addresses 192.168.1.10/24 \
  ipv4.gateway 192.168.1.1 ipv4.dns 192.168.1.1
nmcli con up lab
4 · 重設 root 密碼（rd.break） 
mount -o remount,rw /sysroot
chroot /sysroot
passwd root
touch /.autorelabel
exit && exit
5 · firewalld 永久規則與重載 
firewall-cmd --permanent --add-service=http
firewall-cmd --permanent --add-port=8080/tcp
firewall-cmd --reload
firewall-cmd --list-all
6 · LVM 建立與含檔案系統自動擴充 
pvcreate /dev/sdb1
vgcreate vgdata /dev/sdb1
lvcreate -n lvdata -L 500M vgdata
lvextend -r -L +500M /dev/vgdata/lvdata
